   106→// Estimate byte size of a base64 data URL (approx — 4 chars = 3 bytes, minus header).
   107→function dataUrlBytes(dataUrl: string): number {
   108→  const comma = dataUrl.indexOf(",");
   109→  if (comma < 0) return 0;
   110→  const b64 = dataUrl.slice(comma + 1);
   111→  // base64: 4 chars ≈ 3 bytes
   112→  return Math.floor((b64.length * 3) / 4);
   113→}

   566→  // Handle image file selection → convert to base64 data URL
   567→  // Compress image to max 200KB PNG via canvas. Returns base64 data URL.
   568→  const compressImage = (file: File): Promise<string> => {
   569→    return new Promise((resolve, reject) => {
   570→      const reader = new FileReader();
   571→      reader.onload = () => {
   572→        const img = new window.Image();
   573→        img.onload = () => {
   574→          const MAX_BYTES = 200 * 1024; // 200KB
   575→          // Start with original dimensions (capped to max 1280px on longest side)
   576→          const maxDim = 1280;
   577→          let { width, height } = img;
   578→          if (width > maxDim || height > maxDim) {
   579→            const scale = maxDim / Math.max(width, height);
   580→            width = Math.round(width * scale);
   581→            height = Math.round(height * scale);
   582→          }
   583→          // Try decreasing quality (PNG is lossless, so we reduce dimensions to hit <200KB)
   584→          const tryCompress = (w: number, h: number): string | null => {
   585→            const canvas = document.createElement("canvas");
   586→            canvas.width = w;
   587→            canvas.height = h;
   588→            const ctx = canvas.getContext("2d");
   589→            if (!ctx) return null;
   590→            ctx.drawImage(img, 0, 0, w, h);
   591→            const dataUrl = canvas.toDataURL("image/png");
   592→            return dataUrl;
   593→          };
   594→          let result = tryCompress(width, height);
   595→          // If still > 200KB, progressively reduce dimensions
   596→          let curW = width;
   597→          let curH = height;
   598→          while (result && dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   599→            curW = Math.round(curW * 0.8);
   600→            curH = Math.round(curH * 0.8);
   601→            result = tryCompress(curW, curH);
   602→          }
   603→          resolve(result || "");
   604→        };
   605→        img.onerror = reject;
   606→        img.src = reader.result as string;
   607→      };
   608→      reader.onerror = reject;
   609→      reader.readAsDataURL(file);
   610→    });
   611→  };

   613→  // Compress image for banner/logo — uses JPEG with dynamic quality to hit <200KB.
   614→  // For PNG with transparency (logos), keeps PNG but reduces dimensions.
   615→  const compressBannerImage = (file: File, useJpeg = true): Promise<string> => {
   616→    return new Promise((resolve, reject) => {
   617→      const reader = new FileReader();
   618→      reader.onload = () => {
   619→        const img = new window.Image();
   620→        img.onload = () => {
   621→          const MAX_BYTES = 200 * 1024; // 200KB
   622→          const maxDim = 1280;
   623→          let { width, height } = img;
   624→          if (width > maxDim || height > maxDim) {
   625→            const scale = maxDim / Math.max(width, height);
   626→            width = Math.round(width * scale);
   627→            height = Math.round(height * scale);
   628→          }
   629→          const canvas = document.createElement("canvas");
   630→          canvas.width = width;
   631→          canvas.height = height;
   632→          const ctx = canvas.getContext("2d");
   633→          if (!ctx) { reject(new Error("Canvas not supported")); return; }
   634→          ctx.drawImage(img, 0, 0, width, height);
   635→          if (useJpeg) {
   636→            // JPEG with dynamic quality reduction
   637→            let quality = 0.9;
   638→            let result = canvas.toDataURL("image/jpeg", quality);
   639→            while (dataUrlBytes(result) > MAX_BYTES && quality > 0.1) {
   640→              quality -= 0.1;
   641→              result = canvas.toDataURL("image/jpeg", quality);
   642→            }
   643→            // If still > 200KB, reduce dimensions
   644→            let curW = width, curH = height;
   645→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   646→              curW = Math.round(curW * 0.8);
   647→              curH = Math.round(curH * 0.8);
   648→              canvas.width = curW;
   649→              canvas.height = curH;
   650→              ctx.drawImage(img, 0, 0, curW, curH);
   651→              result = canvas.toDataURL("image/jpeg", quality);
   652→            }
   653→            resolve(result);
   654→          } else {
   655→            // PNG (for logos with transparency)
   656→            let result = canvas.toDataURL("image/png");
   657→            let curW = width, curH = height;
   658→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   659→              curW = Math.round(curW * 0.8);
   660→              curH = Math.round(curH * 0.8);
   661→              canvas.width = curW;
   662→              canvas.height = curH;
   663→              ctx.drawImage(img, 0, 0, curW, curH);
   664→              result = canvas.toDataURL("image/png");
   665→            }
   666→            resolve(result);
   667→          }
   668→        };
   669→        img.onerror = reject;
   670→        img.src = reader.result as string;
   671→      };
   672→      reader.onerror = reject;
   673→      reader.readAsDataURL(file);
   674→    });
