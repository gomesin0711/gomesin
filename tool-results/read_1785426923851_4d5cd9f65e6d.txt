    87→    87→async function postListing(payload: any) {
    88→    88→  const res = await fetch("/api/listings", {
    89→    89→    method: "POST",
    90→    90→    headers: { "Content-Type": "application/json" },
    91→    91→    body: JSON.stringify(payload),
    92→    92→  });
    93→    93→  const data = await res.json();
