    40→// ============ FETCHERS ============
    41→const fetchJson = async (url: string) => {
    42→  const res = await fetch(url);
    43→  if (!res.ok) throw new Error(`API ${res.status}`);
    44→  const text = await res.text();
    45→  if (!text) throw new Error(`API ${res.status}: empty response`);
    46→  return JSON.parse(text);
    47→};
    48→
    49→// Safe admin query — returns empty data on error instead of throwing
    50→function useAdminQuery<T>(key: string[], url: string) {
    51→  return useQuery({
    52→    queryKey: key,
    53→    queryFn: async () => {
    54→      try { return await fetchJson(url); } catch { return null; }
    55→    },
    56→    retry: false,
    57→    staleTime: 0,
    58→  });
    59→}

   251→function getRemainingDaysAdmin(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
   252→  if (!paymentExpiry) return { days: -1, expired: false };
   253→  const now = new Date();
   254→  const exp = new Date(paymentExpiry);
   255→  const diff = exp.getTime() - now.getTime();
   256→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   257→  return { days, expired: days <= 0 };
   258→}

   260→function getPkgBadge(type: string) {
   261→  switch (type) {
   262→    case "spotlight": return { name: "Titanium", bg: "bg-amber-100 text-amber-700", dot: "bg-amber-500" };
   263→    case "highlight": return { name: "Platinum", bg: "bg-orange-100 text-orange-700", dot: "bg-orange-500" };
   264→    case "sundul": return { name: "Colek", bg: "bg-purple-100 text-purple-700", dot: "bg-purple-500" };
   265→    case "colek": return { name: "Gold", bg: "bg-blue-100 text-blue-700", dot: "bg-blue-500" };
   266→    default: return { name: "Gratis", bg: "bg-secondary text-muted-foreground", dot: "bg-muted-foreground" };
