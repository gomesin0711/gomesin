    61→// --- Helpers ---
    62→function getRemainingDays(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
    63→  if (!paymentExpiry) return { days: -1, expired: false };
    64→  const now = new Date();
    65→  const exp = new Date(paymentExpiry);
    66→  const diff = exp.getTime() - now.getTime();
    67→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
    68→  return { days, expired: days <= 0 };
    69→}
    70→
    71→function formatRemainingDays(paymentExpiry: string | null | undefined, lang: string): string {
    72→  const { days, expired } = getRemainingDays(paymentExpiry);
    73→  if (days < 0) return "-";
    74→  if (expired) return "Non Aktif";
    75→  if (days === 0) return "Hari ini berakhir";
    76→  if (days === 1) return "1 hari lagi";
    77→  return `${days} hari lagi`;
    78→}

    80→async function fetchListings(userId?: string) {
    81→  const url = userId ? `/api/my-listings?userId=${encodeURIComponent(userId)}` : "/api/my-listings";
    82→  const res = await fetch(url);
    83→  if (!res.ok) throw new Error("fail");
    84→  return res.json() as Promise<{ listings: Listing[]; total: number }>;
    85→}
    86→
    87→async function deleteListing(slug: string) {
    88→  const res = await fetch(`/api/listings/${slug}`, { method: "DELETE" });
    89→  const data = await res.json();
    90→  if (!res.ok) throw new Error(data.error || "Gagal menghapus");
    91→  return data;
    92→}
