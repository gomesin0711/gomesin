   248→type AdminPkgTabKey = (typeof ADMIN_PKG_TABS)[number]["key"];
   249→
   250→function getRemainingDaysAdmin(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
   251→  if (!paymentExpiry) return { days: -1, expired: false };
   252→  const now = new Date();
   253→  const exp = new Date(paymentExpiry);
   254→  const diff = exp.getTime() - now.getTime();
   255→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   256→  return { days, expired: days <= 0 };
   257→}
   258→
   259→function getPkgBadge(type: string) {
   260→  switch (type) {
   261→    case "spotlight": return { name: "Titanium", bg: "bg-amber-100 text-amber-700", dot: "bg-amber-500" };
   262→    case "highlight": return { name: "Platinum", bg: "bg-orange-100 text-orange-700", dot: "bg-orange-500" };
   263→    case "sundul": return { name: "Colek", bg: "bg-purple-100 text-purple-700", dot: "bg-purple-500" };
   264→    case "colek": return { name: "Gold", bg: "bg-blue-100 text-blue-700", dot: "bg-blue-500" };
   265→    default: return { name: "Gratis", bg: "bg-secondary text-muted-foreground", dot: "bg-muted-foreground" };
   266→  }
   267→}
