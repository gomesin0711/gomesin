   800→  2960→function timeAgoShort(dateStr: string): string {
   801→  2961→  if (!dateStr) return "—";
   802→  2962→  const diff = Date.now() - new Date(dateStr).getTime();
   803→  2963→  const sec = Math.floor(diff / 1000);
   804→  2964→  if (sec < 60) return `${sec}s`;
   805→  2965→  const min = Math.floor(sec / 60);
   806→  2966→  if (min < 60) return `${min}m`;
   807→  2967→  const hr = Math.floor(min / 60);
   808→  2968→  if (hr < 24) return `${hr}j`;
   809→  2969→  const day = Math.floor(hr / 24);
   810→  2970→  if (day < 30) return `${day}h`;
   811→  2971→  const mo = Math.floor(day / 30);
   812→  2972→  return `${mo}bl`;
   813→  2973→}
