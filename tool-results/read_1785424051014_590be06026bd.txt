    93→  open,
    94→  onOpenChange,
    95→  onEdit,
    96→  onSuccess,
    97→}: {
    98→  listing: Listing;
    99→  open: boolean;

   102→  onSuccess?: () => void;
   103→}) {

   105→  const mounted = useMounted();

   107→  const queryClient = useQueryClient();
   108→  const user = useStore((s) => s.user);

   113→    queryFn: async () => {

   145→    };

   150→

   159→  const needsPayment = selectedPackage !== "simpan" && selectedPkg.price > 0;
   160→
   161→  // Status info for listing summary
   162→  const statusInfo =
   163→    listing.status === "pending"

   168→          text: (listing as any).violationFlag ? tr("violation") : tr("rejected"),

   173→
   174→  // Current package label for listing summary
   175→  const currentPkgLabel =
   176→    listing.packageType === "spotlight"

   179→      ? "Platinum"
   180→      : listing.packageType === "sundul"
   181→      ? "Colek"

   190→  const currentUserId = useStore((s) => s.user?.id);

   204→          setUniqueCode(data.uniqueCode);
   205→        }

   209→  }, [open, listing.id, currentUserId, selectedPackage, selectedPkg.price]);

   212→    ? selectedPkg.price + uniqueCode
   213→    : selectedPkg.price;

   217→      toast.error("Pilih metode pembayaran terlebih dahulu.");

   223→      setQrisModal(true);

   227→      setBcaModal(true);

   237→      const res = await fetch(`/api/listings/${listing.slug}`, {
   238→        method: "PATCH",
   239→        headers: { "Content-Type": "application/json" },
   240→        body: JSON.stringify({
   241→          package: selectedPackage,
   242→          paymentMethod: needsPayment ? paymentMethod : undefined,
   243→        }),
   244→      });

   246→      if (!res.ok) throw new Error(data.error || "Gagal mengaktifkan paket");

   253→        onSuccess();

   255→        onOpenChange(false);
   256→      }
   257→    } catch (e: any) {
   258→      toast.error(e.message || "Gagal mengaktifkan paket");
   259→    } finally {
   260→      setSubmitting(false);
   261→    }
   262→  };
   263→

   265→    <>
   266→    {/* ===== UPGRADE PAKET PAGE (fullscreen, 2 kolom di desktop) ===== */}

   268→      <div className="fixed inset-0 z-[60] flex h-screen flex-col overflow-hidden bg-background">

   272→            <div>
   273→              <h1 className="text-xl font-bold sm:text-2xl">Upgrade Paket Iklan</h1>

   275→            </div>
   276→            <button
   277→              type="button"
   278→              onClick={() => onOpenChange(false)}
   279→              className="grid size-10 shrink-0 place-items-center rounded-full border border-border bg-card hover:bg-accent"
   280→              aria-label="Tutup"
   281→            >
   282→              <X className="size-5" />
   283→            </button>
   284→          </div>
   285→
   286→          {/* Content 2 kolom (desktop) / 1 kolom (mobile) */}
   287→          <div className="grid flex-1 gap-6 md:grid-cols-2 md:overflow-hidden">
   288→
   289→            {/* LEFT — gambar iklan + detail */}
   290→            <div className="space-y-3 md:overflow-y-auto md:pr-2">
   291→              <div className="overflow-hidden rounded-xl border border-border bg-card">
   292→                {/* Gambar utama iklan (besar) */}
   293→                <div className="relative aspect-video w-full overflow-hidden bg-muted">
   294→                  {listing.images?.[0] ? (
   295→                    <Image
   296→                      src={listing.images[0]}
   297→                      alt={listingTitle(listing, mounted ? lang : "id")}
   298→                      fill
   299→                      className="object-cover"
   300→                      unoptimized
   301→                      sizes="(max-width: 768px) 100vw, 50vw"
   302→                    />
   303→                  ) : (
   304→                    <div className="flex h-full items-center justify-center text-muted-foreground">
   305→                      <ImageIcon className="size-12" />
   306→                    </div>
   307→                  )}
   308→                </div>
   309→                {/* Detail iklan */}
   310→                <div className="space-y-2 p-4">
   311→                  <p className="line-clamp-2 text-base font-semibold text-foreground">
   312→                    {listingTitle(listing, mounted ? lang : "id")}
   313→                  </p>
   314→                  <p className="text-xl font-bold text-primary">
   315→                    {formatRupiahFull(listing.price)}
   316→                    {listing.priceType === "negotiable" && (
   317→                      <span className="ml-2 text-xs font-medium text-muted-foreground">Nego</span>

   319→                  </p>
   320→                  <div className="flex flex-wrap items-center gap-1.5">
   321→                    <span
   322→                      className={cn(
   323→                        "inline-flex items-center gap-1 rounded px-2 py-0.5 text-[10px] font-bold uppercase text-white",
   324→                        statusInfo.color
   325→                      )}
   326→                    >
   327→                      <StatusIcon className="size-3" />
   328→                      {statusInfo.text}
   329→                    </span>
   330→                    <span className="inline-flex items-center rounded bg-secondary px-2 py-0.5 text-[10px] font-medium text-muted-foreground">
   331→                      Paket saat ini: {currentPkgLabel}
   332→                    </span>
   333→                  </div>
   334→                  {/* Info tambahan */}
   335→                  <div className="grid grid-cols-2 gap-x-3 gap-y-1 pt-2 text-xs">
   336→                    <div className="flex justify-between border-b border-border/50 pb-1">
   337→                      <span className="text-muted-foreground">Kondisi</span>
   338→                      <span className="font-medium">{listing.condition === "baru" ? "Baru" : "Bekas"}</span>

   348→                        <span className="text-muted-foreground">Tahun</span>

   351→                    )}
   352→                    <div className="flex justify-between border-b border-border/50 pb-1">
   353→                      <span className="text-muted-foreground">Lokasi</span>
   354→                      <span className="font-medium">{listing.city}</span>
   355→                    </div>
   356→                  </div>

   366→                    </div>
   367→                  )}

   399→                        p.key === "spotlight"

   507→      {/* ===== QRIS PAYMENT PAGE (overlay fullscreen, sama persis seperti post-ad) ===== */}

   821→/**
   822→ * Sanitize user name/email to be filesystem-safe.
   823→ * "John Doe" → "John_Doe"
   824→ * "user@email.com" → "user_email.com"
   825→ * Removes special chars that break URLs/filenames.
   826→ */
   827→function sanitizeFileName(name: string): string {
   828→  return name
   829→    .trim()
   830→    .replace(/\s+/g, "_")           // spaces → underscore
   831→    .replace(/[@/\\?%*:|"<>]/g, "_") // special chars → underscore
   832→    .replace(/_+/g, "_")              // collapse multiple underscores
   833→    .substring(0, 50);               // max 50 chars
   834→}
