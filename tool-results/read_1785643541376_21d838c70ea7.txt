   105→// Estimate byte size of a base64 data URL (approx — 4 chars = 3 bytes, minus header).
   106→function dataUrlBytes(dataUrl: string): number {
   107→  const comma = dataUrl.indexOf(",");
   108→  if (comma < 0) return 0;
   109→  const b64 = dataUrl.slice(comma + 1);
   110→  // base64: 4 chars ≈ 3 bytes
   111→  return Math.floor((b64.length * 3) / 4);
   112→}

   553→  // Handle image file selection → convert to base64 data URL
   554→  // Compress image to max 200KB PNG via canvas. Returns base64 data URL.
   555→  const compressImage = (file: File): Promise<string> => {
   556→    return new Promise((resolve, reject) => {
   557→      const reader = new FileReader();
   558→      reader.onload = () => {
   559→        const img = new window.Image();
   560→        img.onload = () => {
   561→          const MAX_BYTES = 200 * 1024; // 200KB
   562→          // Start with original dimensions (capped to max 1280px on longest side)
   563→          const maxDim = 1280;
   564→          let { width, height } = img;
   565→          if (width > maxDim || height > maxDim) {
   566→            const scale = maxDim / Math.max(width, height);
   567→            width = Math.round(width * scale);
   568→            height = Math.round(height * scale);
   569→          }
   570→          // Try decreasing quality (PNG is lossless, so we reduce dimensions to hit <200KB)
   571→          const tryCompress = (w: number, h: number): string | null => {
   572→            const canvas = document.createElement("canvas");
   573→            canvas.width = w;
   574→            canvas.height = h;
   575→            const ctx = canvas.getContext("2d");
   576→            if (!ctx) return null;
   577→            ctx.drawImage(img, 0, 0, w, h);
   578→            const dataUrl = canvas.toDataURL("image/png");
   579→            return dataUrl;
   580→          };
   581→          let result = tryCompress(width, height);
   582→          // If still > 200KB, progressively reduce dimensions
   583→          let curW = width;
   584→          let curH = height;
   585→          while (result && dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   586→            curW = Math.round(curW * 0.8);
   587→            curH = Math.round(curH * 0.8);
   588→            result = tryCompress(curW, curH);
   589→          }
   590→          resolve(result || "");
   591→        };
   592→        img.onerror = reject;
   593→        img.src = reader.result as string;
   594→      };
   595→      reader.onerror = reject;
   596→      reader.readAsDataURL(file);
   597→    });
   598→  };

   600→  // Compress image for banner/logo — uses JPEG with dynamic quality to hit <200KB.
   601→  // For PNG with transparency (logos), keeps PNG but reduces dimensions.
   602→  const compressBannerImage = (file: File, useJpeg = true): Promise<string> => {
   603→    return new Promise((resolve, reject) => {
   604→      const reader = new FileReader();
   605→      reader.onload = () => {
   606→        const img = new window.Image();
   607→        img.onload = () => {
   608→          const MAX_BYTES = 200 * 1024; // 200KB
   609→          const maxDim = 1280;
   610→          let { width, height } = img;
   611→          if (width > maxDim || height > maxDim) {
   612→            const scale = maxDim / Math.max(width, height);
   613→            width = Math.round(width * scale);
   614→            height = Math.round(height * scale);
   615→          }
   616→          const canvas = document.createElement("canvas");
   617→          canvas.width = width;
   618→          canvas.height = height;
   619→          const ctx = canvas.getContext("2d");
   620→          if (!ctx) { reject(new Error("Canvas not supported")); return; }
   621→          ctx.drawImage(img, 0, 0, width, height);
   622→          if (useJpeg) {
   623→            // JPEG with dynamic quality reduction
   624→            let quality = 0.9;
   625→            let result = canvas.toDataURL("image/jpeg", quality);
   626→            while (dataUrlBytes(result) > MAX_BYTES && quality > 0.1) {
   627→              quality -= 0.1;
   628→              result = canvas.toDataURL("image/jpeg", quality);
   629→            }
   630→            // If still > 200KB, reduce dimensions
   631→            let curW = width, curH = height;
   632→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   633→              curW = Math.round(curW * 0.8);
   634→              curH = Math.round(curH * 0.8);
   635→              canvas.width = curW;
   636→              canvas.height = curH;
   637→              ctx.drawImage(img, 0, 0, curW, curH);
   638→              result = canvas.toDataURL("image/jpeg", quality);
   639→            }
   640→            resolve(result);
   641→          } else {
   642→            // PNG (for logos with transparency)
   643→            let result = canvas.toDataURL("image/png");
   644→            let curW = width, curH = height;
   645→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   646→              curW = Math.round(curW * 0.8);
   647→              curH = Math.round(curH * 0.8);
   648→              canvas.width = curW;
   649→              canvas.height = curH;
   650→              ctx.drawImage(img, 0, 0, curW, curH);
   651→              result = canvas.toDataURL("image/png");
   652→            }
   653→            resolve(result);
   654→          }
   655→        };
   656→        img.onerror = reject;
   657→        img.src = reader.result as string;
   658→      };
   659→      reader.onerror = reject;
   660→      reader.readAsDataURL(file);
   661→    });
