   105→// Estimate byte size of a base64 data URL (approx — 4 chars = 3 bytes, minus header).
   106→function dataUrlBytes(dataUrl: string): number {
   107→  const comma = dataUrl.indexOf(",");
   108→  if (comma < 0) return 0;
   109→  const b64 = dataUrl.slice(comma + 1);
   110→  // base64: 4 chars ≈ 3 bytes
   111→  return Math.floor((b64.length * 3) / 4);
   112→}

   552→  // Handle image file selection → convert to base64 data URL
   553→  // Compress image to max 200KB PNG via canvas. Returns base64 data URL.
   554→  const compressImage = (file: File): Promise<string> => {
   555→    return new Promise((resolve, reject) => {
   556→      const reader = new FileReader();
   557→      reader.onload = () => {
   558→        const img = new window.Image();
   559→        img.onload = () => {
   560→          const MAX_BYTES = 200 * 1024; // 200KB
   561→          // Start with original dimensions (capped to max 1280px on longest side)
   562→          const maxDim = 1280;
   563→          let { width, height } = img;
   564→          if (width > maxDim || height > maxDim) {
   565→            const scale = maxDim / Math.max(width, height);
   566→            width = Math.round(width * scale);
   567→            height = Math.round(height * scale);
   568→          }
   569→          // Try decreasing quality (PNG is lossless, so we reduce dimensions to hit <200KB)
   570→          const tryCompress = (w: number, h: number): string | null => {
   571→            const canvas = document.createElement("canvas");
   572→            canvas.width = w;
   573→            canvas.height = h;
   574→            const ctx = canvas.getContext("2d");
   575→            if (!ctx) return null;
   576→            ctx.drawImage(img, 0, 0, w, h);
   577→            const dataUrl = canvas.toDataURL("image/png");
   578→            return dataUrl;
   579→          };
   580→          let result = tryCompress(width, height);
   581→          // If still > 200KB, progressively reduce dimensions
   582→          let curW = width;
   583→          let curH = height;
   584→          while (result && dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   585→            curW = Math.round(curW * 0.8);
   586→            curH = Math.round(curH * 0.8);
   587→            result = tryCompress(curW, curH);
   588→          }
   589→          resolve(result || "");
   590→        };
   591→        img.onerror = reject;
   592→        img.src = reader.result as string;
   593→      };
   594→      reader.onerror = reject;
   595→      reader.readAsDataURL(file);
   596→    });
   597→  };

   599→  // Compress image for banner/logo — uses JPEG with dynamic quality to hit <200KB.
   600→  // For PNG with transparency (logos), keeps PNG but reduces dimensions.
   601→  const compressBannerImage = (file: File, useJpeg = true): Promise<string> => {
   602→    return new Promise((resolve, reject) => {
   603→      const reader = new FileReader();
   604→      reader.onload = () => {
   605→        const img = new window.Image();
   606→        img.onload = () => {
   607→          const MAX_BYTES = 200 * 1024; // 200KB
   608→          const maxDim = 1280;
   609→          let { width, height } = img;
   610→          if (width > maxDim || height > maxDim) {
   611→            const scale = maxDim / Math.max(width, height);
   612→            width = Math.round(width * scale);
   613→            height = Math.round(height * scale);
   614→          }
   615→          const canvas = document.createElement("canvas");
   616→          canvas.width = width;
   617→          canvas.height = height;
   618→          const ctx = canvas.getContext("2d");
   619→          if (!ctx) { reject(new Error("Canvas not supported")); return; }
   620→          ctx.drawImage(img, 0, 0, width, height);
   621→          if (useJpeg) {
   622→            // JPEG with dynamic quality reduction
   623→            let quality = 0.9;
   624→            let result = canvas.toDataURL("image/jpeg", quality);
   625→            while (dataUrlBytes(result) > MAX_BYTES && quality > 0.1) {
   626→              quality -= 0.1;
   627→              result = canvas.toDataURL("image/jpeg", quality);
   628→            }
   629→            // If still > 200KB, reduce dimensions
   630→            let curW = width, curH = height;
   631→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   632→              curW = Math.round(curW * 0.8);
   633→              curH = Math.round(curH * 0.8);
   634→              canvas.width = curW;
   635→              canvas.height = curH;
   636→              ctx.drawImage(img, 0, 0, curW, curH);
   637→              result = canvas.toDataURL("image/jpeg", quality);
   638→            }
   639→            resolve(result);
   640→          } else {
   641→            // PNG (for logos with transparency)
   642→            let result = canvas.toDataURL("image/png");
   643→            let curW = width, curH = height;
   644→            while (dataUrlBytes(result) > MAX_BYTES && curW > 100) {
   645→              curW = Math.round(curW * 0.8);
   646→              curH = Math.round(curH * 0.8);
   647→              canvas.width = curW;
   648→              canvas.height = curH;
   649→              ctx.drawImage(img, 0, 0, curW, curH);
   650→              result = canvas.toDataURL("image/png");
   651→            }
   652→            resolve(result);
   653→          }
   654→        };
   655→        img.onerror = reject;
   656→        img.src = reader.result as string;
   657→      };
   658→      reader.onerror = reject;
   659→      reader.readAsDataURL(file);
   660→    });
