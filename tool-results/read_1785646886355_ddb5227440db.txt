   212→type AdminPkgTabKey = (typeof ADMIN_PKG_TABS)[number]["key"];
   213→
   214→function getRemainingDaysAdmin(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
   215→  if (!paymentExpiry) return { days: -1, expired: false };
   216→  const now = new Date();
   217→  const exp = new Date(paymentExpiry);
   218→  const diff = exp.getTime() - now.getTime();
   219→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   220→  return { days, expired: days <= 0 };
   221→}
   222→
   223→function getPkgBadge(type: string) {
   224→  switch (type) {
   225→    case "spotlight": return { name: "Titanium", bg: "bg-amber-100 text-amber-700", dot: "bg-amber-500" };
   226→    case "highlight": return { name: "Platinum", bg: "bg-orange-100 text-orange-700", dot: "bg-orange-500" };
   227→    case "sundul": return { name: "Colek", bg: "bg-purple-100 text-purple-700", dot: "bg-purple-500" };
   228→    case "colek": return { name: "Gold", bg: "bg-blue-100 text-blue-700", dot: "bg-blue-500" };
   229→    default: return { name: "Gratis", bg: "bg-secondary text-muted-foreground", dot: "bg-muted-foreground" };
   230→  }
   231→}
