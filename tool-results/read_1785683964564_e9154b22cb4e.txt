  2980→function timeAgoShort(dateStr: string): string {
  2981→  if (!dateStr) return "—";
  2982→  const diff = Date.now() - new Date(dateStr).getTime();
  2983→  const sec = Math.floor(diff / 1000);
  2984→  if (sec < 60) return `${sec}s`;
  2985→  const min = Math.floor(sec / 60);
  2986→  if (min < 60) return `${min}m`;
  2987→  const hr = Math.floor(min / 60);
  2988→  if (hr < 24) return `${hr}j`;
  2989→  const day = Math.floor(hr / 24);
  2990→  if (day < 30) return `${day}h`;
  2991→  const mo = Math.floor(day / 30);
  2992→  return `${mo}bl`;
  2993→}
