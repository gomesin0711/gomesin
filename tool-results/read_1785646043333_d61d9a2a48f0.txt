   113→// Slug generator
   114→function slugify(text: string): string {
   115→  return text
   116→    .toLowerCase()
   117→    .replace(/[^a-z0-9\u4e00-\u9fa5]+/g, "-")
   118→    .replace(/(^-|-$)/g, "");
   119→}
   120→
   121→function pick<T>(arr: T[], indices: number[]): T[] {
   122→  return indices.map((i) => arr[i % arr.length]);
   123→}

   564→
   565→  // 1. Delete ALL existing listings
   566→  const deleted = await db.listing.deleteMany({});
   567→  console.log(`Deleted ${deleted.count} existing listings`);
   568→
   569→  // 2. Upsert categories
   570→  const catMap: Record<string, string> = {};
   571→  for (const c of categories) {
   572→    const cat = await db.category.upsert({
   573→      where: { slug: c.slug },
   574→      update: { name: c.name, icon: c.icon, color: c.color, sortOrder: c.sortOrder },
   575→      create: c,
   576→    });
   577→    catMap[c.slug] = cat.id;
   578→    console.log(`Category: ${cat.name} (${cat.id})`);
   579→  }
   580→
   581→  // 3. Upsert sellers
   582→  const sellerMap: string[] = [];
   583→  for (const s of sellers) {
   584→    const seller = await db.seller.upsert({
   585→      where: { id: (await db.seller.findFirst({ where: { name: s.name } }))?.id || "" },
   586→      update: s,
   587→      create: s,
   588→    });
   589→    sellerMap.push(seller.id);
   590→    console.log(`Seller: ${seller.name} (${seller.id})`);
   591→  }
   592→
   593→  // 4. Create listings
   594→  let created = 0;
   595→  for (const l of listings) {
   596→    const catId = catMap[l.catSlug];
   597→    if (!catId) {
   598→      console.error(`SKIP: category ${l.catSlug} not found`);
   599→      continue;
   600→    }
   601→    const sellerId = sellerMap[created % sellerMap.length];
   602→    const images = pick(IMG[l.imgKey], l.imgs);
