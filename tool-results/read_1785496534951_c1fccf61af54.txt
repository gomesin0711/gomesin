  2960→function timeAgoShort(dateStr: string): string {
  2961→  if (!dateStr) return "—";
  2962→  const diff = Date.now() - new Date(dateStr).getTime();
  2963→  const sec = Math.floor(diff / 1000);
  2964→  if (sec < 60) return `${sec}s`;
  2965→  const min = Math.floor(sec / 60);
  2966→  if (min < 60) return `${min}m`;
  2967→  const hr = Math.floor(min / 60);
  2968→  if (hr < 24) return `${hr}j`;
  2969→  const day = Math.floor(hr / 24);
  2970→  if (day < 30) return `${day}h`;
  2971→  const mo = Math.floor(day / 30);
  2972→  return `${mo}bl`;
  2973→}
