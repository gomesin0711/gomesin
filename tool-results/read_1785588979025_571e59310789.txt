    47→type OtpState = "idle" | "sending" | "sent" | "verifying" | "verified";
    48→
    49→function useOtp() {
    50→  const [otpState, setOtpState] = useState<OtpState>("idle");
    51→  const [otpCode, setOtpCode] = useState("");
    52→  const [cooldown, setCooldown] = useState(0);
    53→  const timerRef = useRef<ReturnType<typeof setInterval>>();
    54→
    55→  const clearTimer = useCallback(() => {
    56→    if (timerRef.current) {
    57→      clearInterval(timerRef.current);
    58→      timerRef.current = undefined;
    59→    }
    60→  }, []);
    61→
    62→  const sendOtp = useCallback(async (phone: string, tr: (k: any) => any) => {
    63→    if (phone.replace(/\D/g, "").length < 10) {
    64→      toast.error(tr("errPhoneRequired"));
    65→      return;
    66→    }
    67→    setOtpState("sending");
    68→    try {
    69→      const res = await fetch("/api/auth/otp", {
    70→        method: "POST",
    71→        headers: { "Content-Type": "application/json" },
    72→        body: JSON.stringify({ phone, action: "send" }),
    73→      });
    74→      const data = await res.json();
    75→      if (!res.ok) {
    76→        toast.error(data.error || "Gagal mengirim OTP");
    77→        setOtpState("idle");
    78→        return;
    79→      }
    80→      // Dev: show OTP code
    81→      if (data._devCode) {
    82→        toast.info(`[DEV] OTP: ${data._devCode}`);
    83→      }
    84→      setOtpState("sent");
    85→      setOtpCode("");
    86→      setCooldown(60);
    87→      toast.success(tr("otpSent"));
    88→    } catch {
    89→      toast.error(tr("errConnection"));
    90→      setOtpState("idle");
    91→    }
    92→  }, []);
    93→
    94→  const verifyOtp = useCallback(async (phone: string, tr: (k: any) => any) => {
    95→    if (otpCode.length !== 6) {
    96→      toast.error(tr("errOtpRequired"));
    97→      return;
    98→    }
    99→    setOtpState("verifying");
   100→    try {
   101→      const res = await fetch("/api/auth/otp", {
   102→        method: "POST",
   103→        headers: { "Content-Type": "application/json" },
   104→        body: JSON.stringify({ phone, action: "verify", code: otpCode }),
   105→      });
   106→      const data = await res.json();
   107→      if (!res.ok) {
   108→        toast.error(data.error || "Gagal verifikasi OTP");
   109→        setOtpState("sent");
   110→        return;
   111→      }
   112→      setOtpState("verified");
   113→      toast.success(tr("otpVerified"));
   114→    } catch {
   115→      toast.error(tr("errConnection"));
   116→      setOtpState("sent");
   117→    }
   118→  }, [otpCode]);
   119→
   120→  // Cooldown timer
   121→  useEffect(() => {
   122→    if (cooldown <= 0) { clearTimer(); return; }
   123→    timerRef.current = setInterval(() => {
   124→      setCooldown((p) => {
   125→        if (p <= 1) { clearTimer(); return 0; }
   126→        return p - 1;
   127→      });
   128→    }, 1000);
   129→    return clearTimer;
   130→  }, [cooldown, clearTimer]);
   131→
   132→  const reset = useCallback(() => {
   133→    setOtpState("idle");
   134→    setOtpCode("");
   135→    setCooldown(0);
   136→    clearTimer();
   137→  }, [clearTimer]);
   138→
   139→  return { otpState, setOtpState, otpCode, setOtpCode, cooldown, sendOtp, verifyOtp, reset };
   140→}

   145→
   146→function WaOtpField({
   147→  phone, setPhone, otp, tr,
   148→}: {
   149→  phone: string; setPhone: (v: string) => void;
   150→  otp: ReturnType<typeof useOtp>;
   151→  tr: (k: any) => any;
   152→}) {
   153→  const { otpState, otpCode, setOtpCode, cooldown, sendOtp, verifyOtp } = otp;
   154→  const digits = phone.replace(/\D/g, "");
   155→  const prevDigitsRef = useRef("");
   156→
   157→  // Auto-send OTP when digits reach 10+
   158→  useEffect(() => {
   159→    if (digits.length >= 10 && digits.length > prevDigitsRef.current.length && otpState === "idle") {
   160→      prevDigitsRef.current = digits;
   161→      sendOtp(phone, tr);
   162→    } else {
   163→      prevDigitsRef.current = digits;
   164→    }
   165→  }, [digits, otpState, phone, sendOtp, tr]);
   166→
   167→  // Auto-verify when 6 digits entered
   168→  useEffect(() => {
   169→    if (otpCode.length === 6 && (otpState === "sent" || otpState === "verifying")) {
   170→      verifyOtp(phone, tr);
   171→    }
   172→  }, [otpCode, otpState, phone, verifyOtp, tr]);
   173→
   174→  return (
   175→    <div className="space-y-3">
   176→      <div className="space-y-1.5">
   177→        <Label htmlFor="wa-phone">{tr("whatsapp")}</Label>
   178→        <div className="flex gap-2">
   179→          <div className="relative flex-1">
   180→            <Phone className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   181→            <Input
   182→              id="wa-phone"
   183→              type="tel"
   184→              inputMode="numeric"
   185→              value={phone}
   186→              onChange={(e) => {
   187→                setPhone(e.target.value.replace(/[^0-9+\- ]/g, ""));
   188→                if (otpState === "verified") otp.reset();
   189→              }}
   190→              placeholder={tr("whatsappPlaceholder")}
   191→              className="pl-9"
   192→            />
   193→          </div>
   194→          <Button
   195→            type="button"
   196→            variant="outline"
   197→            size="sm"
   198→            disabled={digits.length < 10 || otpState === "sending" || cooldown > 0}
   199→            onClick={() => sendOtp(phone, tr)}
   200→            className="shrink-0 gap-1.5 px-3"
   201→          >
   202→            {otpState === "sending" ? (
   203→              <Loader2 className="size-3.5 animate-spin" />
   204→            ) : cooldown > 0 ? (
   205→              <>{cooldown}s</>
   206→            ) : otpState === "sent" || otpState === "verified" ? (
   207→              <RefreshCw className="size-3.5" />
   208→            ) : (
   209→              <Send className="size-3.5" />
   210→            )}
   211→            {cooldown > 0
   212→              ? tr("resendOtp")
   213→              : otpState === "sent" || otpState === "verified"
   214→                ? tr("resendOtp")
   215→                : tr("sendOtp")}
   216→          </Button>
   217→        </div>
   218→      </div>
   219→
   220→      {(otpState === "sent" || otpState === "verifying" || otpState === "verified") && (
   221→        <div className="space-y-1.5 animate-fade-up">
   222→          <Label htmlFor="otp-code">{tr("enterOtp")}</Label>
   223→          <div className="relative">
   224→            <Input
   225→              id="otp-code"
   226→              type="text"
   227→              inputMode="numeric"
   228→              maxLength={6}
   229→              value={otpCode}
   230→              onChange={(e) => setOtpCode(e.target.value.replace(/[^0-9]/g, "").slice(0, 6))}
   231→              placeholder="000000"
   232→              className={cn(
   233→                "pl-9 tracking-[0.3em] text-center font-mono text-lg",
   234→                otpState === "verified" && "border-green-500 bg-green-50 dark:bg-green-950/30"
   235→              )}
   236→              disabled={otpState === "verifying"}
   237→            />
   238→            {otpState === "verifying" && (
   239→              <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 size-4 animate-spin text-primary" />
   240→            )}
   241→            {otpState === "verified" && (
   242→              <CheckCircle2 className="absolute right-3 top-1/2 -translate-y-1/2 size-4 text-green-500" />
   243→            )}
   244→          </div>
   245→        </div>
   246→      )}
   247→    </div>
   248→  );
   249→}

   268→  const [lEmail, setLEmail] = useState("");
   269→  const [lPass, setLPass] = useState("");
   270→  const [lPhone, setLPhone] = useState("");
   271→  const loginOtp = useOtp();
   272→

   275→  const [rEmail, setREmail] = useState("");
   276→  const [rPhone, setRPhone] = useState("");

   297→        if (!lEmail.trim() || !lPass) {
   298→          toast.error(tr("errEmailPass"));
   299→          return;

   303→        if (lPhone.replace(/\D/g, "").length < 10) {
   304→          toast.error(tr("errPhoneRequired"));

   308→          toast.error(tr("errPhoneNotVerified"));
   309→          return;

   313→      const res = await fetch("/api/auth/login", {
   314→        method: "POST",
   315→        headers: { "Content-Type": "application/json" },
   316→        body: JSON.stringify(body),
   317→      });
   318→      const data = await res.json();

   320→        toast.error(data.error || tr("errLogin"));
   321→        return;

   343→    }

   345→      toast.error(tr("errPhoneRequired"));
   346→      return;
   347→    }
   348→    if (registerOtp.otpState !== "verified") {
   349→      toast.error(tr("errPhoneNotVerified"));
   350→      return;
   351→    }
   352→    if (rPass.length < 6) {
   353→      toast.error(tr("errPassLength"));
   354→      return;
   355→    }
   356→    if (rPass !== rPass2) {
   357→      toast.error(tr("errPassMatch"));
   358→      return;
   359→    }
   360→    if (!agree) {
   361→      toast.error(tr("errAgree"));

   368→        headers: { "Content-Type": "application/json" },

   374→          city: rCity || rProvince,
