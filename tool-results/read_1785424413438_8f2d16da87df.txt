   631→      <div className="mb-4 overflow-x-auto gomesin-scroll">

   672→            />
   673→          </div>

   676→        <div className="flex items-center gap-2">
   677→          {/* View mode toggle */}
   678→          <div className="flex overflow-hidden rounded-lg border border-border">
   679→            <button
   680→              type="button"
   681→              onClick={() => setViewMode("grid")}

   684→                "grid size-9 place-items-center transition",

   692→              onClick={() => setViewMode("line")}

   694→              className={cn(
   695→                "grid size-9 place-items-center border-l border-border transition",
   696→                viewMode === "line" ? "bg-primary text-primary-foreground" : "bg-background text-foreground hover:bg-accent"
   697→              )}
   698→            >
   699→              <List className="size-4" />
   700→            </button>
   701→          </div>
   702→        </div>
   703→      </div>
   704→
   705→      {/* ===== RESULTS ===== */}
   706→      {isLoading ? (
   707→        viewMode === "grid" ? (
   708→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   709→            {Array.from({ length: 8 }).map((_, i) => (

   715→                  <div className="h-3 w-1/2 animate-pulse rounded bg-muted" />
   716→                </div>
   717→              </div>
   718→            ))}
   719→          </div>
   720→        ) : (
   721→          <div className="rounded-xl border border-border bg-card">
   722→            {Array.from({ length: 6 }).map((_, i) => (
   723→              <div key={i} className="flex gap-3 border-b border-border p-3">
   724→                <div className="h-20 w-20 shrink-0 animate-pulse rounded-lg bg-muted sm:h-24 sm:w-24" />

   732→          </div>
   733→        )

   735→        <div className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border py-16 text-center">
   736→          <Frown className="size-12 text-muted-foreground" />
   737→          <h3 className="mt-3 text-lg font-semibold">{tr("noAds")}</h3>
   738→          <p className="mt-1 max-w-sm text-sm text-muted-foreground">{tr("noAdsDesc")}</p>
   739→          <Button className="mt-4 gap-2" onClick={goToPost}>
   740→            <Plus className="size-4" /> {tr("postFirstAd")}
   741→          </Button>
   742→        </div>
   743→      ) : viewMode === "grid" ? (
   744→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   745→          {listings.map(renderGridCard)}
   746→        </div>
   747→      ) : (
   748→        <div className="rounded-xl border border-border bg-card overflow-hidden">
   749→          {listings.map(renderLineCard)}
   750→        </div>
   751→      )}
   752→
   753→      {/* delete confirmation dialog */}
   754→      <AlertDialog open={deleteSlug !== null} onOpenChange={(o) => !o && setDeleteSlug(null)}>
   755→        <AlertDialogContent>
   756→          <AlertDialogHeader>
   757→            <AlertDialogTitle>{tr("deleteAdTitle")}</AlertDialogTitle>
   758→            <AlertDialogDescription>{tr("deleteAdDesc")}</AlertDialogDescription>
   759→          </AlertDialogHeader>
   760→          <AlertDialogFooter>
   761→            <AlertDialogCancel>{tr("cancel")}</AlertDialogCancel>
   762→            <AlertDialogAction
   763→              onClick={confirmDelete}
   764→              disabled={deleteMutation.isPending}
   765→              className="gap-2 bg-destructive text-white hover:bg-destructive/90"
   766→            >
   767→              {deleteMutation.isPending ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   768→              {deleteMutation.isPending ? tr("deleting") : tr("deleteBtn")}
   769→            </AlertDialogAction>
   770→          </AlertDialogFooter>
   771→        </AlertDialogContent>
   772→      </AlertDialog>
   773→    </div>
   774→  );
