   220→function getRemainingDaysAdmin(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
   221→  if (!paymentExpiry) return { days: -1, expired: false };
   222→  const now = new Date();
   223→  const exp = new Date(paymentExpiry);
   224→  const diff = exp.getTime() - now.getTime();
   225→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   226→  return { days, expired: days <= 0 };
   227→}

   229→function getPkgBadge(type: string) {
   230→  switch (type) {
   231→    case "spotlight": return { name: "Titanium", bg: "bg-amber-100 text-amber-700", dot: "bg-amber-500" };
   232→    case "highlight": return { name: "Platinum", bg: "bg-orange-100 text-orange-700", dot: "bg-orange-500" };
   233→    case "sundul": return { name: "Colek", bg: "bg-purple-100 text-purple-700", dot: "bg-purple-500" };
   234→    case "colek": return { name: "Gold", bg: "bg-blue-100 text-blue-700", dot: "bg-blue-500" };
   235→    default: return { name: "Gratis", bg: "bg-secondary text-muted-foreground", dot: "bg-muted-foreground" };
