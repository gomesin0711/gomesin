   769→  2960→function timeAgoShort(dateStr: string): string {
   770→  2961→  if (!dateStr) return "—";
   771→  2962→  const diff = Date.now() - new Date(dateStr).getTime();
   772→  2963→  const sec = Math.floor(diff / 1000);
   773→  2964→  if (sec < 60) return `${sec}s`;
   774→  2965→  const min = Math.floor(sec / 60);
   775→  2966→  if (min < 60) return `${min}m`;
   776→  2967→  const hr = Math.floor(min / 60);
   777→  2968→  if (hr < 24) return `${hr}j`;
   778→  2969→  const day = Math.floor(hr / 24);
   779→  2970→  if (day < 30) return `${day}h`;
   780→  2971→  const mo = Math.floor(day / 30);
   781→  2972→  return `${mo}bl`;
   782→  2973→}
