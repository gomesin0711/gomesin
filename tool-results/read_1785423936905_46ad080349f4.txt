    76→async function fetchCategories() {
    77→  const res = await fetch("/api/categories");
    78→  if (!res.ok) throw new Error("fail");
    79→  return (await res.json()).categories as Array<{
    80→    id: string;
    81→    name: string;
    82→    slug: string;
    83→    icon: string;
    84→  }>;
    85→}
    86→
    87→async function postListing(payload: any) {
    88→  const res = await fetch("/api/listings", {
    89→    method: "POST",
    90→    headers: { "Content-Type": "application/json" },
    91→    body: JSON.stringify(payload),
    92→  });
    93→  const data = await res.json();
    94→  if (!res.ok) throw new Error(data.error || tr("postFailed"));
    95→  return data.listing;
    96→}
