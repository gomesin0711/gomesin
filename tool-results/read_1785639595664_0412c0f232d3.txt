   260→type AdminPkgTabKey = (typeof ADMIN_PKG_TABS)[number]["key"];
   261→
   262→function getRemainingDaysAdmin(paymentExpiry: string | null | undefined): { days: number; expired: boolean } {
   263→  if (!paymentExpiry) return { days: -1, expired: false };
   264→  const now = new Date();
   265→  const exp = new Date(paymentExpiry);
   266→  const diff = exp.getTime() - now.getTime();
   267→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   268→  return { days, expired: days <= 0 };
   269→}
   270→
   271→function getPkgBadge(type: string) {
   272→  switch (type) {
   273→    case "spotlight": return { name: "Titanium", bg: "bg-amber-100 text-amber-700", dot: "bg-amber-500" };
   274→    case "highlight": return { name: "Platinum", bg: "bg-orange-100 text-orange-700", dot: "bg-orange-500" };
   275→    case "sundul": return { name: "Colek", bg: "bg-purple-100 text-purple-700", dot: "bg-purple-500" };
   276→    case "colek": return { name: "Gold", bg: "bg-blue-100 text-blue-700", dot: "bg-blue-500" };
   277→    default: return { name: "Gratis", bg: "bg-secondary text-muted-foreground", dot: "bg-muted-foreground" };
   278→  }
   279→}

   707→  };
   708→  const reject = (id: string) => {
   709→    if (confirm(tr("admRejectConfirm"))) {
   710→      setStatus.mutate({ id, status: "rejected" });
   711→    }
   712→  };
   713→
   714→  const openPreview = (l: any) => { setPreviewListing(l); setActiveImg(0); };

   719→    const imgs = l.images || [];
   720→    const pkg = getPkgBadge(l.packageType);
   721→
   722→    return (
   723→      <div
   724→        key={l.id}
   725→        onClick={() => openPreview(l)}
   726→        className="group flex cursor-pointer flex-col overflow-hidden rounded-xl border-2 border-amber-300 bg-card transition hover:shadow-lg"

   728→        <div className="relative aspect-[4/3] w-full overflow-hidden bg-muted">
