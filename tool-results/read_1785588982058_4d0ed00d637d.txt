    47→    47→type OtpState = "idle" | "sending" | "sent" | "verifying" | "verified";
    48→    48→
    49→    49→function useOtp() {
    50→    50→  const [otpState, setOtpState] = useState<OtpState>("idle");
    51→    51→  const [otpCode, setOtpCode] = useState("");
    52→    52→  const [cooldown, setCooldown] = useState(0);
    53→    53→  const timerRef = useRef<ReturnType<typeof setInterval>>();
    54→    54→
    55→    55→  const clearTimer = useCallback(() => {
    56→    56→    if (timerRef.current) {
    57→    57→      clearInterval(timerRef.current);
    58→    58→      timerRef.current = undefined;
    59→    59→    }
    60→    60→  }, []);
    61→    61→
    62→    62→  const sendOtp = useCallback(async (phone: string, tr: (k: any) => any) => {
    63→    63→    if (phone.replace(/\D/g, "").length < 10) {
    64→    64→      toast.error(tr("errPhoneRequired"));
    65→    65→      return;
    66→    66→    }
    67→    67→    setOtpState("sending");
    68→    68→    try {
    69→    69→      const res = await fetch("/api/auth/otp", {
    70→    70→        method: "POST",
    71→    71→        headers: { "Content-Type": "application/json" },
    72→    72→        body: JSON.stringify({ phone, action: "send" }),
    73→    73→      });
    74→    74→      const data = await res.json();
    75→    75→      if (!res.ok) {
    76→    76→        toast.error(data.error || "Gagal mengirim OTP");
    77→    77→        setOtpState("idle");
    78→    78→        return;
    79→    79→      }
    80→    80→      // Dev: show OTP code
    81→    81→      if (data._devCode) {
    82→    82→        toast.info(`[DEV] OTP: ${data._devCode}`);
    83→    83→      }
    84→    84→      setOtpState("sent");
    85→    85→      setOtpCode("");
    86→    86→      setCooldown(60);
    87→    87→      toast.success(tr("otpSent"));
    88→    88→    } catch {
    89→    89→      toast.error(tr("errConnection"));
    90→    90→      setOtpState("idle");
    91→    91→    }
    92→    92→  }, []);
    93→    93→
    94→    94→  const verifyOtp = useCallback(async (phone: string, tr: (k: any) => any) => {
    95→    95→    if (otpCode.length !== 6) {
    96→    96→      toast.error(tr("errOtpRequired"));
    97→    97→      return;
    98→    98→    }
    99→    99→    setOtpState("verifying");
   100→   100→    try {
   101→   101→      const res = await fetch("/api/auth/otp", {
   102→   102→        method: "POST",
   103→   103→        headers: { "Content-Type": "application/json" },
   104→   104→        body: JSON.stringify({ phone, action: "verify", code: otpCode }),
   105→   105→      });
   106→   106→      const data = await res.json();
   107→   107→      if (!res.ok) {
   108→   108→        toast.error(data.error || "Gagal verifikasi OTP");
   109→   109→        setOtpState("sent");
   110→   110→        return;
   111→   111→      }
   112→   112→      setOtpState("verified");
   113→   113→      toast.success(tr("otpVerified"));
   114→   114→    } catch {
   115→   115→      toast.error(tr("errConnection"));
   116→   116→      setOtpState("sent");
   117→   117→    }
   118→   118→  }, [otpCode]);
   119→   119→
   120→   120→  // Cooldown timer
   121→   121→  useEffect(() => {
   122→   122→    if (cooldown <= 0) { clearTimer(); return; }
   123→   123→    timerRef.current = setInterval(() => {
   124→   124→      setCooldown((p) => {
   125→   125→        if (p <= 1) { clearTimer(); return 0; }
   126→   126→        return p - 1;
   127→   127→      });
   128→   128→    }, 1000);
   129→   129→    return clearTimer;
   130→   130→  }, [cooldown, clearTimer]);
   131→   131→
   132→   132→  const reset = useCallback(() => {
   133→   133→    setOtpState("idle");
   134→   134→    setOtpCode("");
   135→   135→    setCooldown(0);
   136→   136→    clearTimer();
   137→   137→  }, [clearTimer]);
   138→   138→
   139→   139→  return { otpState, setOtpState, otpCode, setOtpCode, cooldown, sendOtp, verifyOtp, reset };
   140→   140→}

   143→   143→/*  WhatsApp OTP Input Field                                          */

   145→   145→
   146→   146→function WaOtpField({
   147→   147→  phone, setPhone, otp, tr,
   148→   148→}: {
   149→   149→  phone: string; setPhone: (v: string) => void;
   150→   150→  otp: ReturnType<typeof useOtp>;
   151→   151→  tr: (k: any) => any;
   152→   152→}) {
   153→   153→  const { otpState, otpCode, setOtpCode, cooldown, sendOtp, verifyOtp } = otp;
   154→   154→  const digits = phone.replace(/\D/g, "");
   155→   155→  const prevDigitsRef = useRef("");
   156→   156→
   157→   157→  // Auto-send OTP when digits reach 10+
   158→   158→  useEffect(() => {
   159→   159→    if (digits.length >= 10 && digits.length > prevDigitsRef.current.length && otpState === "idle") {
   160→   160→      prevDigitsRef.current = digits;
   161→   161→      sendOtp(phone, tr);
   162→   162→    } else {
   163→   163→      prevDigitsRef.current = digits;
   164→   164→    }
   165→   165→  }, [digits, otpState, phone, sendOtp, tr]);
   166→   166→
   167→   167→  // Auto-verify when 6 digits entered
   168→   168→  useEffect(() => {
   169→   169→    if (otpCode.length === 6 && (otpState === "sent" || otpState === "verifying")) {
   170→   170→      verifyOtp(phone, tr);
   171→   171→    }
   172→   172→  }, [otpCode, otpState, phone, verifyOtp, tr]);
   173→   173→
   174→   174→  return (
   175→   175→    <div className="space-y-3">
   176→   176→      <div className="space-y-1.5">
   177→   177→        <Label htmlFor="wa-phone">{tr("whatsapp")}</Label>
   178→   178→        <div className="flex gap-2">
   179→   179→          <div className="relative flex-1">
   180→   180→            <Phone className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   181→   181→            <Input
   182→   182→              id="wa-phone"
   183→   183→              type="tel"
   184→   184→              inputMode="numeric"
   185→   185→              value={phone}
   186→   186→              onChange={(e) => {
   187→   187→                setPhone(e.target.value.replace(/[^0-9+\- ]/g, ""));
   188→   188→                if (otpState === "verified") otp.reset();
   189→   189→              }}
   190→   190→              placeholder={tr("whatsappPlaceholder")}
   191→   191→              className="pl-9"
   192→   192→            />
   193→   193→          </div>
   194→   194→          <Button
   195→   195→            type="button"
   196→   196→            variant="outline"
   197→   197→            size="sm"
   198→   198→            disabled={digits.length < 10 || otpState === "sending" || cooldown > 0}
   199→   199→            onClick={() => sendOtp(phone, tr)}
   200→   200→            className="shrink-0 gap-1.5 px-3"
   201→   201→          >
   202→   202→            {otpState === "sending" ? (
   203→   203→              <Loader2 className="size-3.5 animate-spin" />
   204→   204→            ) : cooldown > 0 ? (
   205→   205→              <>{cooldown}s</>
   206→   206→            ) : otpState === "sent" || otpState === "verified" ? (
   207→   207→              <RefreshCw className="size-3.5" />
   208→   208→            ) : (
   209→   209→              <Send className="size-3.5" />
   210→   210→            )}
   211→   211→            {cooldown > 0
   212→   212→              ? tr("resendOtp")
   213→   213→              : otpState === "sent" || otpState === "verified"
   214→   214→                ? tr("resendOtp")
   215→   215→                : tr("sendOtp")}
   216→   216→          </Button>
   217→   217→        </div>
   218→   218→      </div>
   219→   219→
   220→   220→      {(otpState === "sent" || otpState === "verifying" || otpState === "verified") && (
   221→   221→        <div className="space-y-1.5 animate-fade-up">
   222→   222→          <Label htmlFor="otp-code">{tr("enterOtp")}</Label>
   223→   223→          <div className="relative">
   224→   224→            <Input
   225→   225→              id="otp-code"
   226→   226→              type="text"
   227→   227→              inputMode="numeric"
   228→   228→              maxLength={6}
   229→   229→              value={otpCode}
   230→   230→              onChange={(e) => setOtpCode(e.target.value.replace(/[^0-9]/g, "").slice(0, 6))}
   231→   231→              placeholder="000000"
   232→   232→              className={cn(
   233→   233→                "pl-9 tracking-[0.3em] text-center font-mono text-lg",
   234→   234→                otpState === "verified" && "border-green-500 bg-green-50 dark:bg-green-950/30"
   235→   235→              )}
   236→   236→              disabled={otpState === "verifying"}
   237→   237→            />
   238→   238→            {otpState === "verifying" && (
   239→   239→              <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 size-4 animate-spin text-primary" />
   240→   240→            )}
   241→   241→            {otpState === "verified" && (
   242→   242→              <CheckCircle2 className="absolute right-3 top-1/2 -translate-y-1/2 size-4 text-green-500" />
   243→   243→            )}
   244→   244→          </div>
   245→   245→        </div>
   246→   246→      )}
   247→   247→    </div>
   248→   248→  );
   249→   249→}
